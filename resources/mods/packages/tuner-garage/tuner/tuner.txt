# The tuner-garage mod's data file: the part catalog and the per-car profiles.
#
# Read with the game's own overlay rules: the file inside the enabled
# `tuner-garage` package (or a loose <mods>/tuner/tuner.txt) shadows this one,
# so a user can add a car or a part without touching the repo. Editing this
# file needs a game restart (the catalog is read once).
#
# Format (INI-like, '#' comments):
#
#   [part:<id>]            a visual part. <id> is the save's part id: renaming
#                          one orphans the saved installs (they are skipped on
#                          load, never lost - they come back if the id does).
#     name = ...           the menu label
#     slot = spoiler       the attach slot: spoiler | exhaust
#     price = 9500         what the garage charges [INF: no retail pricing
#                          table exists; the tuning shop's levels are $12k steps]
#     mesh = wing_gt.obj   the OBJ in this folder's parts/ (Blender -> OBJ+MTL,
#                          the mod-props route). Y=0 is the deck plane the part
#                          sits on, +Z points at the car's nose (the car body
#                          space is rear = -Z; measured on the GTR at runtime),
#                          metres.
#     offset = 0 1.02 -1.9 the default body-space position (x y z, metres):
#                          where the part's origin lands on the car body node.
#     node = BODY_HR       the car node the part hangs from (BODY_HR, BODY_MR;
#                          default BODY_HR)
#     align = rear         optional. The offset is then relative to THIS car's
#                          body: x as written, y above the body bottom, z past
#                          the rear face (so -0.03 = 3 cm beyond the bumper).
#                          Exhaust tips use it - one entry fits every car. The
#                          tip mesh's opening is at z=0 facing the rear, the
#                          pipe running +Z into the car.
#
#   [car:<file name>]      a tuner car, keyed by the DBCarPhysicsData
#                          _car_file_name (so both catalog rows of a car - e.g.
#                          the two GTR rows - match). A car absent from this
#                          list is not a tuner car.
#     slots = spoiler, exhaust   the slots its garage menu offers
#     spoiler_offset = ... overrides the slot's default offset for this car
#
# The retail roster is small: no R34, Evo, Supra, RX-7, S15 or NSX exists in
# the game's 279 car rows [DATA: -dumpdbtable DBCarPhysicsData] - those would
# be new car models, a project of their own. The starter list is what fits the
# style today; adding a car is one block here.

[car:GTR]
slots = spoiler, exhaust
# The GTR body carries its own boot-lip spoiler (BODY_HR geometry): without
# the override the low lips (ducktail / street) tuck under it and half
# disappear. Eyeballed against the stock lip (-logtuner bounds: z
# -2.317..+2.182, deck y ~0.85); z keeps the ducktail's 0.26 m depth on the
# deck (-2.05 - 0.26 = -2.31, just inside the body).
spoiler_offset = 0 0.82 -2.05
# The stock parts a fitted part replaces. The car's own EXHAUST node (the stock
# tips) is hidden automatically while an exhaust part is fitted. A stock
# spoiler that is part of the body mesh needs a box: hide_spoiler = x0 y0 z0
# x1 y1 z1 (item space, repeat the key for more boxes); the triangles whose
# centre is inside are dropped while a spoiler is fitted. Find a box with
# -dumptunermask (every rear triangle centre) and -logtuner (how many a box
# catches). Note the model's X is mirrored on screen.
hide_spoiler = -0.95 0.94 -2.30 0.95 1.05 -1.68
